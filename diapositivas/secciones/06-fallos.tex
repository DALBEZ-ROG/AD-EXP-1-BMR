\begin{frame}{Tolerancia a fallos: detectar y recuperarse}
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{block}{Cómo se detecta}
\begin{itemize}
\item Se vence el tiempo de espera (\emph{timeout}).
\item Llega un código HTTP 5xx.
\item Llega un SOAP Fault, como el de nuestro PFC cuando la solicitud no existe.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{Cómo se recupera}
\begin{itemize}
\item Reintentar con esperas cada vez más largas, \alert{solo} si el método es idempotente (\texttt{GET}, \texttt{PUT}, \texttt{DELETE}) \cite{rfc9110}.
\item Tener un plan B, como el respaldo de SGIRI cuando falla Globalping.
\item Analizar la causa raíz después del incidente \cite{soldani2023}.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
